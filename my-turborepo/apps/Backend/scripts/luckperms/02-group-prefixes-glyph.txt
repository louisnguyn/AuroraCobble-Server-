# Rank prefixes using the Asteryn rank pack bitmap font.
# Codepoints from README_rank_pack.txt / assets/minecraft/font/default.json.
# `\uXXXX` is expanded by rcon-batch.ts before sending.
#
# &f before the glyph keeps icon colours. Trailing space separates icon from name.
# Website uses `lp user <name> parent set <role>` (one primary group).
#
# Run: npx tsx scripts/rcon-batch.ts scripts/luckperms/02-group-prefixes-glyph.txt
